VAR _PM =
CALCULATE(
    SELECTEDMEASURE(),
    FILTER( 'dim-Date', 'dim-Date'[MonthnYear] = [Max Previous Month Not Blank] )
)

VAR _CM =
CALCULATE(
    SELECTEDMEASURE(),
    FILTER( 'dim-Date', 'dim-Date'[MonthnYear] = [Max Month Not Blank] )
)

VAR _MOM= DIVIDE( _CM - _PM , _PM )

VAR _CF = IF( _MOM < 0, "#ff383f", "#15b2ad" )
VAR _CF2 = IF( _MOM < 0, "#D60007", "#0F7572" )


VAR _Result =
SWITCH(
    TRUE(),
    CONTAINSSTRINGEXACT( SELECTEDMEASURENAME(), "Actual" ), _CM,
    CONTAINSSTRINGEXACT( SELECTEDMEASURENAME(), "Previous" ), _PM,
    CONTAINSSTRINGEXACT( SELECTEDMEASURENAME(), "Change" ), _MOM,
    CONTAINSSTRINGEXACT( SELECTEDMEASURENAME(), "CF" ), _CF,
    CONTAINSSTRINGEXACT( SELECTEDMEASURENAME(), "color" ), _CF2,
    SELECTEDMEASURE()
)

RETURN  _Result
Depends on: Max Month Not Blank, Max Previous Month Not Blank, dim-Date[MonthnYear]
